    <script type="module">

import {
    auth
} from "./firebase-config.js";

import {
    signInWithEmailAndPassword,
    sendEmailVerification,
    signOut
} from "https://www.gstatic.com/firebasejs/11.0.2/firebase-auth.js";

import { claimSession } from "./session-limit.js";


// =====================================
// LOGIN ELEMENTS
// =====================================

const loginForm =
    document.getElementById("loginForm");

const continueButton =
    document.getElementById("continueButton");

const usernameInput =
    document.getElementById("username");

const passwordInput =
    document.getElementById("password");

const loginMessage =
    document.getElementById("loginMessage");

const showPassword =
    document.getElementById("showPassword");


// =====================================
// SHOW / HIDE PASSWORD
// =====================================

showPassword.addEventListener(
    "click",
    function () {

        if (passwordInput.type === "password") {

            passwordInput.type = "text";

            showPassword.textContent = "Hide";

        } else {

            passwordInput.type = "password";

            showPassword.textContent = "Show";
        }

    }
);


// =====================================
// CONTINUE BUTTON
// =====================================

continueButton.addEventListener(
    "click",
    async function () {
                if (continueButton.dataset.ready === "login") {
            return;
        }

        const email =
            usernameInput.value.trim();

        const password =
            passwordInput.value;


        if (!email || !password) {

            loginMessage.textContent =
                "Enter your email and password first.";

            loginMessage.style.color =
                "#c62828";

            return;
        }


        try {

            // Sign in to check the account

            const userCredential =
                await signInWithEmailAndPassword(
                    auth,
                    email,
                    password
                );


            const user =
                userCredential.user;


            // =====================================
            // ALREADY VERIFIED
            // =====================================

            if (user.emailVerified) {

                loginMessage.textContent =
                    "Email already verified. You can now log in.";

                loginMessage.style.color =
                    "#2e7d32";

                continueButton.textContent =
                    "Login";

                continueButton.disabled =
                    false;

                continueButton.dataset.ready =
                    "login";

                return;
            }


            // =====================================
            // NOT VERIFIED
            // =====================================

            await sendEmailVerification(user);


            loginMessage.textContent =
                "Verification email sent. Check your inbox and click the verification link.";

            loginMessage.style.color =
                "#2e7d32";


            continueButton.textContent =
                "Waiting for Verification...";


            continueButton.disabled =
                true;


            // =====================================
            // CHECK FOR VERIFICATION
            // =====================================

            const verificationCheck =
                setInterval(
                    async function () {

                        try {

                            if (!auth.currentUser) {
                                return;
                            }


                            await auth.currentUser.reload();


                            if (
                                auth.currentUser.emailVerified
                            ) {

                                clearInterval(
                                    verificationCheck
                                );


                                loginMessage.textContent =
                                    "Email verified successfully. You can now log in.";

                                loginMessage.style.color =
                                    "#2e7d32";


                                continueButton.textContent =
                                    "Login";


                                continueButton.disabled =
                                    false;


                                continueButton.dataset.ready =
                                    "login";
                            }

                        } catch (error) {

                            console.error(
                                "Verification check error:",
                                error
                            );

                        }

                    },
                    2000
                );


        } catch (error) {

            console.error(
                "Authentication error:",
                error
            );


            loginMessage.textContent =
                "✕ Incorrect email or password.";

            loginMessage.style.color =
                "#c62828";


            passwordInput.value = "";

            passwordInput.focus();

        }

    }
);


// =====================================
// LOGIN FORM
// =====================================

continueButton.addEventListener(
    "click",
    async function () {

        if (
            continueButton.dataset.ready !== "login"
        ) {
            return;
        }


        const email =
            usernameInput.value.trim();


        const password =
            passwordInput.value;


        try {

            await signInWithEmailAndPassword(
                auth,
                email,
                password
            );


            if (!auth.currentUser.emailVerified) {

                loginMessage.textContent =
                    "Please verify your email before logging in.";

                loginMessage.style.color =
                    "#c62828";

                return;
            }
                     // Check the 4-session limit
            const session = await claimSession();

            if (session.reason === "limit") {
                await signOut(auth);

                loginMessage.textContent =
                    "Maximum login limit reached (4 users). Log out on another device and try again.";

                loginMessage.style.color = "#c62828";
                return;
            }

            // If the check itself failed (network or quota), let the user in
            // instead of locking them out.

            // Save login status

            sessionStorage.setItem(
                "loggedIn",
                "true"
            );


            sessionStorage.setItem(
                "username",
                email
            );


            loginMessage.textContent =
                "✓ Login successful. Opening dashboard...";


            loginMessage.style.color =
                "#2e7d32";


            continueButton.disabled =
                true;


            continueButton.textContent =
                "Opening Dashboard...";


            setTimeout(
                function () {

                    window.location.href =
                        "dashboard.html";

                },
                500
            );


        } catch (error) {

            console.error(
                "Login error:",
                error
            );


            loginMessage.textContent =
                "✕ Incorrect email or password.";

            loginMessage.style.color =
                "#c62828";


            passwordInput.value = "";

            passwordInput.focus();

        }

    }
);


    </script>

function refreshSensorDisplay() {

    const tempEl = document.getElementById("tempStat");
    const humEl = document.getElementById("humStat");

    const connected =
        lastEsp32ReadingAt !== null &&
        Date.now() - lastEsp32ReadingAt <= 90000;

    // ESP32 not sending: show nothing instead of the last stored values
    if (!connected || !latestData) {
        moistureStat.textContent = "--";
        soilCondition.textContent = "Waiting for ESP32";
        soilCondition.style.color = "#89958e";

        if (tempEl) tempEl.textContent = "--";
        if (humEl) humEl.textContent = "--";
        return;
    }

    // ESP32 is sending: show the live values
       
    const moisture = Number(latestData.soilMoisture);

    if (
        latestData.soilMoisture !== undefined &&
        latestData.soilMoisture !== null &&
        Number.isFinite(moisture) &&
        moisture >= 0 &&
        moisture <= 100
    ) {
        moistureStat.textContent = moisture + "%";
        updateSoilCondition(moisture);
    } else {
        // Reading missing or out of range: don't leave an old value showing
        moistureStat.textContent = "--";
        soilCondition.textContent = "Unknown";
        soilCondition.style.color = "#89958e";
    }

    const hasTemp =
        latestData.temperature !== undefined &&
        latestData.temperature !== null &&
        Number.isFinite(Number(latestData.temperature));

    const hasHum =
        latestData.humidity !== undefined &&
        latestData.humidity !== null &&
        Number.isFinite(Number(latestData.humidity));

    if (tempEl) {
        tempEl.textContent = hasTemp ? Number(latestData.temperature) + "°C" : "--";
    }

    if (humEl) {
        humEl.textContent = hasHum ? Number(latestData.humidity) + "%" : "--";
    }
}



function updateESP32Status() {
    if (!deviceConnectionText || !deviceConnectionDot) {
        return;
    }

    const now = Date.now();

    const ageMs = lastEsp32ReadingAt === null
        ? Infinity
        : now - lastEsp32ReadingAt;

    const hasReading = lastEsp32ReadingAt !== null;

        const connected = ageMs <= 90000;
    if (connected) {
        deviceConnectionText.textContent = "ESP32 Connected";
        deviceConnectionDot.style.backgroundColor = "#2e7d32";
    } else if (hasReading) {
        deviceConnectionText.textContent = "ESP32 Offline";
        deviceConnectionDot.style.backgroundColor = "#c62828";
    } else {
        deviceConnectionText.textContent = "Waiting for ESP32...";
        deviceConnectionDot.style.backgroundColor = "#89958e";
    }
}

function updateSoilCondition(moisture) {

    let condition;
    let color;

    if (moisture < DRY_BELOW) {        // 0-29 %
        condition = "Dry";
        color = "#dc2626";
    } else if (moisture < WET_FROM) {  // 30-79 %
        condition = "Moist";
        color = "#b7791f";
    } else {                           // 80-100 %
        condition = "Wet";
        color = "#2e7d32";
    }

    // Top statistics card
    if (soilCondition) {
        soilCondition.textContent = condition;
        soilCondition.style.color = color;
    }
}


// =====================================
// RAIN FORECAST CARD
// rainChance, rainSkip and rainTest are written by the ESP32
// =====================================

function updateRain(data) {

    const icon = document.getElementById("rainIcon");
    const chanceText = document.getElementById("rainChanceText");
    const statusText = document.getElementById("rainStatusText");

    if (!icon || !chanceText || !statusText) {
        return;
    }

    // Demo: test mode pretends rain is coming
    if (data.rainTest === true) {
        icon.textContent = "🧪";
        chanceText.textContent = "Test mode";
        chanceText.style.color = "#1E6FA8";
        statusText.textContent = "Pretending rain: AUTO and SCHEDULE watering skipped";
        return;
    }

    const chance = Number(data.rainChance);

    // -1 or missing = the ESP32 has no forecast yet
    if (!Number.isFinite(chance) || chance < 0) {
        icon.textContent = "☁️";
        chanceText.textContent = "--";
        chanceText.style.color = "#69756e";
        statusText.textContent = "Waiting for forecast";
        return;
    }

    chanceText.textContent = chance + "% chance of rain";

    if (data.rainSkip === true) {
        icon.textContent = "🌧️";
        chanceText.style.color = "#1E6FA8";
        statusText.textContent = "Rain expected: AUTO and SCHEDULE watering skipped";
    } else {
        icon.textContent = "☀️";
        chanceText.style.color = "#2E7D32";
        statusText.textContent = "No rain expected: watering as normal";
    }
}

// =====================================
// LOAD RECENT READINGS
// =====================================

 function loadRecentReadings() {

    const readingsRef = query(
        collection(db, "irrigationReadings"),
        orderBy("timestamp", "desc"),
        limit(4)
    );

    onSnapshot(
        readingsRef,

        function(snapshot) {

            const readings = [];

            snapshot.forEach(function(docSnapshot) {
                readings.push({
                    id: docSnapshot.id,
                    ...docSnapshot.data()
                });
            });

            readings.sort(function(a, b) {
                const timeA = a.timestamp?.toMillis?.() || 0;
                const timeB = b.timestamp?.toMillis?.() || 0;

                return timeB - timeA;
            });

            const recent = readings.slice(0, 4);

            recentReadingsBody.innerHTML = "";

            if (recent.length === 0) {
                recentReadingsBody.innerHTML = `
                    <tr>
                        <td colspan="4">No readings available.</td>
                    </tr>
                `;
                return;
            }

            recent.forEach(function(reading) {

                const moisture = Number(reading.soilMoisture);

                let condition;
                let badgeClass;

                      if (!Number.isFinite(moisture)) {
                    condition = "Unknown";
                    badgeClass = "badge-grey";
                } else if (moisture < DRY_BELOW) {   // Dry: 0-29 %
                    condition = "Dry";
                    badgeClass = "badge-grey";
                } else if (moisture < WET_FROM) {    // Moist: 30-79 %
                    condition = "Moist";
                    badgeClass = "badge-green";
                } else {                             // Wet: 80-100 %
                    condition = "Wet";
                    badgeClass = "badge-green";
                }

                                let time = "Waiting for time";

                if (reading.timestamp?.toDate) {
                    time = reading.timestamp.toDate().toLocaleString(
                        [],
                        {
                            day: "2-digit",
                            month: "short",
                            hour: "2-digit",
                            minute: "2-digit"
                        }
                    );
                }

                const row = document.createElement("tr");

                const timeCell = document.createElement("td");
                timeCell.textContent = time;

                const moistureCell = document.createElement("td");
                moistureCell.textContent =
                    Number.isFinite(moisture) ? moisture + "%" : "Invalid";

                const conditionCell = document.createElement("td");
                const badge = document.createElement("span");

                badge.className = "badge " + badgeClass;
                badge.textContent = condition;
                conditionCell.appendChild(badge);

                const pumpCell = document.createElement("td");
                pumpCell.textContent = reading.pump === true ? "ON" : "OFF";

                row.appendChild(timeCell);
                row.appendChild(moistureCell);
                row.appendChild(conditionCell);
                row.appendChild(pumpCell);

                recentReadingsBody.appendChild(row);
            });
        },

        function(error) {
            console.error("Error loading recent readings:", error);

            recentReadingsBody.innerHTML = `
                <tr>
                    <td colspan="4">Unable to load readings.</td>
                </tr>
            `;
        }
    );
}

// =====================================
// COUNTDOWN TIMER (website side)
// The ESP32 does the real timing; the website
// just shows a countdown so you can see it.
// =====================================

let timerEndTime = null;          // when the countdown should end (ms)
let timerDisplayInterval = null;  // the 1-second display updater


// Shows "Watering... 0:45 left" once per second
function startTimerDisplay(seconds) {

    timerEndTime = Date.now() + seconds * 1000;

    if (timerDisplayInterval) {
        clearInterval(timerDisplayInterval);
    }

    timerDisplayInterval = setInterval(function() {

        const left = Math.max(0, Math.round((timerEndTime - Date.now()) / 1000));
        const mins = Math.floor(left / 60);
        const secs = String(left % 60).padStart(2, "0");

        timerStatus.textContent = "Watering... " + mins + ":" + secs + " left";

        if (left === 0) {
            clearInterval(timerDisplayInterval);
            timerDisplayInterval = null;
            timerStatus.textContent = "Countdown finished";
        }

    }, 1000);
}


// =====================================
// START COUNTDOWN BUTTON
// =====================================

startTimerButton.addEventListener("click", async function() {

    // Countdown only works in TIMER mode
    if (systemMode !== "TIMER") {
        alert("Please select TIMER mode first.");
        return;
    }

    const seconds = Number(timerDuration.value);

    // Same limits as the ESP32 and the Firestore rules
    if (!Number.isInteger(seconds) || seconds < 5 || seconds > 299) {
        alert("Duration must be a whole number between 5 and 299 seconds.");
        return;
    }

    try {
        // Ask the ESP32 to run the pump for this many seconds
        await updateDoc(irrigationRef, {
            timerSeconds: seconds,
            timerRunning: true
        });

        startTimerDisplay(seconds);

    } catch (error) {
        console.error("Unable to start countdown:", error);
        timerStatus.textContent = "Could not start countdown";
    }
});

onSnapshot(
    irrigationRef,

    function(documentSnapshot) {

        if (!documentSnapshot.exists()) {
            console.error("Irrigation document not found.");
            return;
        }

        const data = documentSnapshot.data();

        // Remember when the ESP32 last wrote (drives the "Connected" badge)
        if (data.timestamp && typeof data.timestamp.toDate === "function") {
            const timestampMs = data.timestamp.toDate().getTime();
            if (Number.isFinite(timestampMs)) {
                lastEsp32ReadingAt = timestampMs;
            }
        }
        updateESP32Status();

        // Real pump state, written by the ESP32
        pumpIsOn = data.pump === true;

        
            // Accept all four modes (anything else falls back to AUTO)
        systemMode = ["AUTO", "MANUAL", "TIMER", "SCHEDULE"].includes(data.mode)
            ? data.mode
            : "AUTO";

               updatePump();
        updateModeDisplay();

                // ---------- Rain forecast card ----------
        updateRain(data);

        // ---------- Timer status ----------
        // The ESP32 sets timerRunning = false when the time is up
        if (data.timerRunning !== true && timerDisplayInterval) {
            clearInterval(timerDisplayInterval);
            timerDisplayInterval = null;
            timerEndTime = null;
            timerStatus.textContent = "Countdown finished";
        }

        // ---------- Schedule status ----------
        const dayNames = ["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"];

        if (data.scheduleEnabled === true && data.scheduleTime) {

            // "DAILY" -> "every day", "1,3,5" -> "Mon, Wed, Fri"
            const days = data.scheduleDays === "DAILY"
                ? "every day"
                : String(data.scheduleDays || "")
                    .split(",")
                    .map(function(d) { return dayNames[Number(d)] || "?"; })
                    .join(", ");

            if (data.scheduleRunning === true) {
                scheduleStatus.textContent = "Scheduled watering is running now";
            } else {
                scheduleStatus.textContent =
                    "Schedule ON: " + data.scheduleTime +
                    " for " + data.scheduleDurationSeconds + " s, " + days;
            }

        } else {
            scheduleStatus.textContent = "Schedule is off";
        }

        // Moisture, temperature and humidity are drawn in one place
        latestData = data;
        refreshSensorDisplay();
    },

    function(error) {
        console.error("Error listening to Firestore:", error);
    }
);

function drawMoistureChart(readings) {
    const svg = document.getElementById("moistureChart");
    const message = document.getElementById("moistureChartMessage");
    const line = document.getElementById("moistureLine");
    const area = document.getElementById("moistureArea");
    const pointsGroup = document.getElementById("moisturePoints");
    const labels = document.getElementById("moistureTimeLabels");

    if (!svg || !message || !line || !area || !pointsGroup || !labels) {
        console.error("Moisture chart HTML elements are missing.");
        return;
    }

    // Keep only readings with valid moisture and timestamp data.
    const validReadings = readings.filter(function (reading) {
        return (
            typeof reading.soilMoisture === "number" &&
            reading.soilMoisture >= 0 &&
            reading.soilMoisture <= 100 &&
            reading.timestamp &&
            typeof reading.timestamp.toDate === "function"
        );
    });

    if (validReadings.length === 0) {
        svg.style.display = "none";
        message.style.display = "flex";
        message.textContent = "No moisture readings available yet.";
        labels.innerHTML = "";
        return;
    }

    // Oldest reading on the left, newest on the right.
    const chartReadings = validReadings.slice(-30);
    const chartWidth = 800;
    const chartHeight = 250;

    const coordinates = chartReadings.map(function (reading, index) {
        const x = chartReadings.length === 1
            ? chartWidth / 2
            : index * chartWidth / (chartReadings.length - 1);

        const moisture = reading.soilMoisture;
        const y = chartHeight - (moisture / 100) * chartHeight;

        return {
            x: x,
            y: y,
            moisture: moisture,
            date: reading.timestamp.toDate()
        };
    });

    const linePoints = coordinates.map(function (point) {
        return `${point.x},${point.y}`;
    }).join(" ");

    line.setAttribute("points", linePoints);

    const firstPoint = coordinates[0];
    const lastPoint = coordinates[coordinates.length - 1];

    area.setAttribute(
        "points",
        `${firstPoint.x},250 ${linePoints} ${lastPoint.x},250`
    );

    // Draw a marker for each reading.
    pointsGroup.innerHTML = "";

    coordinates.forEach(function (point) {
        const circle = document.createElementNS(
            "http://www.w3.org/2000/svg",
            "circle"
        );

        circle.setAttribute("cx", point.x);
        circle.setAttribute("cy", point.y);
        circle.setAttribute("r", "4");
        circle.setAttribute("fill", "#2e7d32");

        const tooltip = document.createElementNS(
            "http://www.w3.org/2000/svg",
            "title"
        );

        tooltip.textContent =
            `${point.moisture}% — ${point.date.toLocaleString()}`;

        circle.appendChild(tooltip);
        pointsGroup.appendChild(circle);
    });

    // Show up to six actual reading times.
    labels.innerHTML = "";

    const labelCount = Math.min(6, coordinates.length);

    for (let i = 0; i < labelCount; i++) {
        const index = labelCount === 1
            ? 0
            : Math.round(
                i * (coordinates.length - 1) / (labelCount - 1)
            );

        const label = document.createElement("span");

                label.textContent = coordinates[index].date.toLocaleString(
            [],
            { day: "2-digit", month: "short", hour: "2-digit", minute: "2-digit" }
        );

        labels.appendChild(label);
    }

    message.style.display = "none";
    svg.style.display = "block";
}

 // LOAD HISTORY FROM FIRESTORE

async function loadHistory() {
    const readingsTable = document.getElementById("readingsTable");

    document.getElementById("averageMoisture").textContent = "...";
    document.getElementById("lowestMoisture").textContent = "...";
    document.getElementById("highestMoisture").textContent = "...";
    document.getElementById("pumpActivations").textContent = "...";

    const readingsQuery = query(
        collection(db, "irrigationReadings"),
        orderBy("timestamp", "desc")
    );

    // Listen for live changes from Firebase.
    onSnapshot(readingsQuery, function(readingsSnapshot) {
        const readings = [];

        readingsSnapshot.forEach(function(documentSnapshot) {
            readings.push({
                ...documentSnapshot.data(),
                id: documentSnapshot.id
            });
        });

        // Process readings from oldest to newest.
        readings.reverse();

        // Use the computer's local date for today's summary.
        const today = new Date();
        const todayKey =
            today.getFullYear() + "/" +
            String(today.getMonth() + 1).padStart(2, "0") + "/" +
            String(today.getDate()).padStart(2, "0");

        let totalMoisture = 0;
        let readingCount = 0;
        let highestMoisture = null;
        let lowestMoisture = null;
        let pumpActivations = 0;
        let previousPumpState = null;

        readingsTable.innerHTML = "";

        readings.forEach(function(data) {
            const rawMoisture = data.soilMoisture;
            const moisture = Number(rawMoisture);
            const pump = data.pump;
            const timestamp = data.timestamp;

            let readingDate = null;
            let dateText = "Unknown";
            let timeText = "Unknown";

            if (timestamp && typeof timestamp.toDate === "function") {
                readingDate = timestamp.toDate();

                dateText = readingDate.toLocaleDateString("en-ZA", {
    day: "2-digit",
    month: "short",
    year: "numeric"
});

                timeText = readingDate.toLocaleTimeString([], {
                    hour: "2-digit",
                    minute: "2-digit"
                });
            }

            const dateKey = readingDate
    ? readingDate.getFullYear() + "/" +
      String(readingDate.getMonth() + 1).padStart(2, "0") + "/" +
      String(readingDate.getDate()).padStart(2, "0")
    : "Unknown";

const isToday = dateKey === todayKey;

            // Count pump activations only within today's readings.
            if (isToday && typeof pump === "boolean") {
                if (previousPumpState === false && pump === true) {
                    pumpActivations++;
                }
                previousPumpState = pump;
            } else if (!isToday) {
                previousPumpState = null;
            }

            // Ignore missing or invalid moisture values.
            const hasMoisture =
                rawMoisture !== null &&
                rawMoisture !== undefined &&
                rawMoisture !== "" &&
                Number.isFinite(moisture) &&
                moisture >= 0 &&
                moisture <= 100;

            if (isToday && hasMoisture) {
                totalMoisture += moisture;
                readingCount++;

                if (highestMoisture === null || moisture > highestMoisture) {
                    highestMoisture = moisture;
                }

                if (lowestMoisture === null || moisture < lowestMoisture) {
                    lowestMoisture = moisture;
                }
            }

            // Keep valid readings from every date in the history table.
            if (!hasMoisture) return;

            let condition;
            if (moisture < 40) {
                condition = "Dry";
            } else if (moisture < 70) {
                condition = "Moist";
            } else {
                condition = "Wet";
            }

            const row = document.createElement("tr");
            row.dataset.date = dateKey;
            row.dataset.condition = condition;
            row.dataset.pump = pump === true ? "ON" : "OFF";
                        if (data.savedOffline === true) row.className = "offline-row";

            row.innerHTML = `
                <td>${dateText}</td>
                <td>${timeText}</td>
                <td>${moisture}%</td>
                <td>${condition}</td>
                <td>${pump === true ? "ON" : "OFF"}</td>
                               <td>${data.mode || "AUTO"}${data.savedOffline === true ? ' <span class="offline-tag">Saved offline</span>' : ""}</td>
            `;

                       readingsTable.insertBefore(row, readingsTable.firstChild);
        });

        // Update the summary cards using TODAY'S readings only.
        document.getElementById("averageMoisture").textContent =
            readingCount > 0
                ? (totalMoisture / readingCount).toFixed(1) + "%"
                : "No data today";

        document.getElementById("lowestMoisture").textContent =
            lowestMoisture !== null ? lowestMoisture + "%" : "No data today";

        document.getElementById("highestMoisture").textContent =
            highestMoisture !== null ? highestMoisture + "%" : "No data today";

        document.getElementById("pumpActivations").textContent =
            pumpActivations;

        // The table still contains previous days' records.
        document.getElementById("readingCount").textContent =
            readings.length + " readings";

        // Keep the graph based on the complete history.
        drawMoistureChart(readings);

        // Reapply the selected filters after new data arrives.
        filterReadings();

    }, function(error) {
        console.error("Error loading history from Firebase:", error);

        document.getElementById("averageMoisture").textContent = "Error";
        document.getElementById("lowestMoisture").textContent = "Error";
        document.getElementById("highestMoisture").textContent = "Error";
        document.getElementById("pumpActivations").textContent = "Error";
    });
}

// FILTER READINGS
function filterReadings() {
    let date = document.getElementById("dateFilter").value;

    if (date !== "") {
        date = date.replaceAll("-", "/");
    }

    const condition = document.getElementById("conditionFilter").value;
    const pump = document.getElementById("pumpFilter").value;
    const rows = document.querySelectorAll("#readingsTable tr");
    let visibleRows = 0;

    rows.forEach(function(row) {
        const showDate = date === "" || row.dataset.date === date;
        const showCondition =
            condition === "all" || row.dataset.condition === condition;
        const showPump = pump === "all" || row.dataset.pump === pump;
        const show = showDate && showCondition && showPump;

        row.style.display = show ? "" : "none";
        if (show) visibleRows++;
    });

    document.getElementById("readingCount").textContent =
        visibleRows + " readings";
}
window.filterReadings = filterReadings;

// RESET FILTERS
function resetFilters() {
    document.getElementById("dateFilter").value = "";
    document.getElementById("conditionFilter").value = "all";
    document.getElementById("pumpFilter").value = "all";

    document.querySelectorAll("#readingsTable tr").forEach(function(row) {
        row.style.display = "";
    });

    document.getElementById("readingCount").textContent =
        document.querySelectorAll("#readingsTable tr").length + " readings";
}
window.resetFilters = resetFilters;

function getReadingDate(reading) {

/* DRAW GRAPH USING REAL READINGS */
function drawMoistureChart(readings) {
    const area = document.getElementById("reportMoistureArea");
    const line = document.getElementById("reportMoistureLine");
    const pointsGroup = document.getElementById("reportMoisturePoints");
    const labelsGroup = document.getElementById("reportMoistureLabels");
    const message = document.getElementById("reportChartMessage");

    area.setAttribute("points", "");
    line.setAttribute("points", "");
    pointsGroup.replaceChildren();
    labelsGroup.replaceChildren();

    // Use readings that have both moisture and a valid timestamp.
    // The database query returns newest first, so reverse a copy.
    const chartReadings = readings
        .filter(r =>
            typeof r.soilMoisture === "number" &&
            r.soilMoisture >= 0 &&
            r.soilMoisture <= 100 &&
            getReadingDate(r) !== null
        )
        .slice(0, 30)
        .reverse();

    if (chartReadings.length === 0) {
        message.textContent = "No timestamped readings available to display.";
        return;
    }

    message.textContent =
        "Showing " + chartReadings.length + " recorded readings.";

    const left = 60;
    const right = 660;
    const top = 30;
    const bottom = 230;

    const coordinates = chartReadings.map((reading, index) => {
        const x = chartReadings.length === 1
            ? (left + right) / 2
            : left + index * (right - left) / (chartReadings.length - 1);

        const y = bottom -
            (reading.soilMoisture / 100) * (bottom - top);

        return {
            x,
            y,
            reading,
            date: getReadingDate(reading)
        };
    });

    const linePoints = coordinates
        .map(point => `${point.x},${point.y}`)
        .join(" ");

    line.setAttribute("points", linePoints);

    const areaPoints = [
        `${coordinates[0].x},${bottom}`,
        ...coordinates.map(point => `${point.x},${point.y}`),
        `${coordinates[coordinates.length - 1].x},${bottom}`
    ].join(" ");

    area.setAttribute("points", areaPoints);

    const svgNS = "http://www.w3.org/2000/svg";

    coordinates.forEach((point, index) => {
        const circle = document.createElementNS(svgNS, "circle");
        circle.setAttribute("cx", point.x);
        circle.setAttribute("cy", point.y);
        circle.setAttribute("r", "4");
        circle.setAttribute("fill", "#2f7d50");

        const tooltip = document.createElementNS(svgNS, "title");
        tooltip.textContent =
            `${formatDate(point.date)} ${formatTime(point.date)}: ${point.reading.soilMoisture}%`;

        circle.appendChild(tooltip);
        pointsGroup.appendChild(circle);

        // Show up to five time labels to keep the graph readable.
        const labelCount = Math.min(5, coordinates.length);
        const shouldLabel =
            index === 0 ||
            index === coordinates.length - 1 ||
            (labelCount > 2 &&
                index % Math.max(
                    1,
                    Math.ceil((coordinates.length - 1) / (labelCount - 1))
                ) === 0);

        if (shouldLabel) {
            const label = document.createElementNS(svgNS, "text");
            label.setAttribute("x", point.x);
            label.setAttribute("y", "250");
            label.setAttribute("font-size", "11");
            label.setAttribute("text-anchor", "middle");
            label.textContent = formatTime(point.date);
            labelsGroup.appendChild(label);
        }
    });
}

function drawReadingsTable(readings) {

    });
}

/* DOWNLOAD A REPORT BUILT FROM FIREBASE DATA */
function downloadReport() {
    if (reportReadings.length === 0) {
        alert("There are no recorded readings to download yet.");
        return;
    }

    const moistureValues = reportReadings.map(r => r.soilMoisture);
    const average = moistureValues.reduce((sum, value) => sum + value, 0)
        / moistureValues.length;

    const dates = reportReadings
        .map(getReadingDate)
        .filter(date => date !== null);

    const reportingPeriod = dates.length
        ? `${formatDate(dates[dates.length - 1])} to ${formatDate(dates[0])}`
        : "Dates unavailable";

    let pumpActivations = 0;

    // Readings are newest first. Count OFF -> ON transitions
    // between an older reading and the newer reading.
    for (let i = 0; i < reportReadings.length - 1; i++) {
        if (
            reportReadings[i].pump === true &&
            reportReadings[i + 1].pump === false
        ) {
            pumpActivations++;
        }
    }

    const rows = reportReadings.map(reading => {
        const date = getReadingDate(reading);
        const condition = getCondition(reading.soilMoisture).text;
        const pump = getPumpStatus(reading).text;

        return `${formatDate(date)} | ${formatTime(date)} | ${reading.soilMoisture.toFixed(1)}% | ${condition} | ${pump} | ${reading.mode || "--"}`;
    });

    const report = `
SMART IRRIGATION AND REAL-TIME MONITORING SYSTEM
SYSTEM REPORT

Reporting period: ${reportingPeriod}

--------------------------------------------

MOISTURE SUMMARY

Average moisture: ${average.toFixed(1)}%
Highest moisture: ${Math.max(...moistureValues).toFixed(1)}%
Lowest moisture: ${Math.min(...moistureValues).toFixed(1)}%
Pump activations: ${pumpActivations}

--------------------------------------------

RECORDED READINGS

Date | Time | Moisture | Condition | Pump | Mode
${rows.join("\n")}

--------------------------------------------

Report generated by Smart Irrigation Dashboard.
`;

    const blob = new Blob([report], { type: "text/plain;charset=utf-8" });
    const url = URL.createObjectURL(blob);
    const link = document.createElement("a");

    link.href = url;
    link.download = "Smart_Irrigation_Report.txt";
    document.body.appendChild(link);
    link.click();
    link.remove();

    URL.revokeObjectURL(url);
}

window.downloadReport = downloadReport;

        const moistureValues = reportReadings.map(r => r.soilMoisture);

    }
}

// session-limit.js
// Allows one account to be logged in on at most 4 devices/tabs at the same time.
// Put this file in the same folder as firebase-config.js.

import { db, auth } from "./firebase-config.js";

import {
    doc,
    runTransaction
} from "https://www.gstatic.com/firebasejs/11.0.2/firebase-firestore.js";

    signOut

export const MAX_SESSIONS = 4;

// A session that has sent no heartbeat for this long is treated as closed.
const STALE_MS = 90000;      // 90 seconds

// How often an open page tells Firestore "I am still here".
const BEAT_MS = 30000;       // 30 seconds

export const LIMIT_MESSAGE =
    "Maximum login limit reached (4 users). " +
    "Log out on another device and try again.";


// Each browser tab gets its own session id.
function getSessionId() {
    let id = sessionStorage.getItem("sessionId");

    if (!id) {
        id = "s" +
            Date.now().toString(36) +
            Math.random().toString(36).slice(2, 10);

        sessionStorage.setItem("sessionId", id);
    }

    return id;
}


// Claims a slot, or refreshes the slot this tab already has.
// Returns { ok: true } or { ok: false, reason: "limit" | "error" | "not-signed-in" }
export async function claimSession() {
    const user = auth.currentUser;

    if (!user) {
        return { ok: false, reason: "not-signed-in" };
    }

    const ref = doc(db, "activeSessions", user.uid);
    const id = getSessionId();

    try {
        return await runTransaction(db, async function (tx) {
            const snap = await tx.get(ref);
            const now = Date.now();

            const saved = snap.exists()
                ? (snap.data().sessions || {})
                : {};

            // Keep only OTHER sessions that are still alive.
            const active = {};

            Object.keys(saved).forEach(function (key) {
                const lastSeen = saved[key];

                if (
                    key !== id &&
                    typeof lastSeen === "number" &&
                    now - lastSeen <= STALE_MS
                ) {
                    active[key] = lastSeen;
                }
            });

            if (Object.keys(active).length >= MAX_SESSIONS) {
                return { ok: false, reason: "limit" };
            }

            active[id] = now;
            tx.set(ref, { sessions: active });

            return { ok: true };
        });

    } catch (error) {
        console.error("Session check failed:", error);
        return { ok: false, reason: "error" };
    }
}


// Frees this tab's slot (call this when the user logs out).
export async function releaseSession() {
    const user = auth.currentUser;
    const id = sessionStorage.getItem("sessionId");

    if (!user || !id) {
        return;
    }

    const ref = doc(db, "activeSessions", user.uid);

    try {
        await runTransaction(db, async function (tx) {
            const snap = await tx.get(ref);

            if (!snap.exists()) {
                return;
            }

            const sessions = Object.assign({}, snap.data().sessions || {});
            delete sessions[id];

            tx.set(ref, { sessions: sessions });
        });

    } catch (error) {
        console.error("Could not release session:", error);
    }

    sessionStorage.removeItem("sessionId");
}


// Call once on every protected page (dashboard, history, reports).
// Keeps this tab's slot alive. If the slot was lost and the limit is
// full, the user is signed out and sent back to the login page.
export function startSessionHeartbeat() {
    if (sessionStorage.getItem("loggedIn") !== "true") {
        return;
    }

    let timer = null;

    onAuthStateChanged(auth, function (user) {
        if (timer) {
            clearInterval(timer);
            timer = null;
        }

        if (!user) {
            return;
        }

        async function beat() {
            const result = await claimSession();

            if (!result.ok && result.reason === "limit") {
                clearInterval(timer);

                sessionStorage.removeItem("loggedIn");
                sessionStorage.removeItem("username");
                sessionStorage.removeItem("sessionId");

                await signOut(auth);

                alert(LIMIT_MESSAGE);
                window.location.href = "index.html";
            }
        }

        beat();
        timer = setInterval(beat, BEAT_MS);
    });
}
